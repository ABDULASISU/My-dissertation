// =============================================================================
// halo2-bench/src/main.rs
//
// Benchmark PLONK/HALO2 sur BN254 avec ~10^6 contraintes de multiplication.
// Fork PSE (git main) — KZG + SHPLONK + Blake2b sur BN254.
// =============================================================================

use std::{
    env,
    time::{Duration, Instant},
};

// ── High-level Imports via halo2_proofs ──────────────────────────────────────
use halo2_proofs::{
    circuit::{Layouter, SimpleFloorPlanner, Value},
    plonk::{create_proof, keygen_pk, keygen_vk, Advice, Circuit, Column, ConstraintSystem, Selector},
    poly::{
        commitment::ParamsProver,
        kzg::{
            commitment::{KZGCommitmentScheme, ParamsKZG},
            multiopen::{ProverSHPLONK, VerifierSHPLONK},
            strategy::SingleStrategy,
        },
        Rotation, // Placé au bon endroit de l'API umbrella
    },
    transcript::{
        Blake2bRead, Blake2bWrite, Challenge255,
        TranscriptReadBuffer, TranscriptWriterBuffer,
    },
};

// ── Chemin explicite vers le verifier du Backend ─────────────────────────────
use halo2_backend::plonk::verifier::verify_proof;

// ── Éléments requis provenant du Frontend ────────────────────────────────────
use halo2_frontend::plonk::Error as FrontendError;

// Courbe BN254 via le re-export PSE
use halo2_proofs::halo2curves::bn256::{Bn256, Fr, G1Affine};
use halo2_proofs::halo2curves::ff::Field;

use rand::rngs::OsRng;

// ── Constantes ────────────────────────────────────────────────────────────────
const DEFAULT_K: u32  = 20;
const BLINDING_ROWS: usize = 6; 

// ── Config PLONK ──────────────────────────────────────────────────────────────
#[derive(Clone, Debug)]
struct MulConfig {
    x:     Column<Advice>,
    x_sq:  Column<Advice>,
    s_mul: Selector,
}

// ── Circuit ───────────────────────────────────────────────────────────────────
#[derive(Clone, Debug, Default)]
struct MulBenchCircuit {
    xs: Vec<Value<Fr>>,
}

impl Circuit<Fr> for MulBenchCircuit {
    type Config       = MulConfig;
    type FloorPlanner = SimpleFloorPlanner;

    fn without_witnesses(&self) -> Self {
        Self { xs: vec![Value::unknown(); self.xs.len()] }
    }

    fn configure(meta: &mut ConstraintSystem<Fr>) -> MulConfig {
        let x     = meta.advice_column();
        let x_sq  = meta.advice_column();
        let s_mul = meta.selector();

        meta.create_gate("squaring", |vc| {
            let xe   = vc.query_advice(x,    Rotation::cur());
            let xsqe = vc.query_advice(x_sq, Rotation::cur());
            let s    = vc.query_selector(s_mul);
            vec![s * (xsqe - xe.clone() * xe)]
        });

        MulConfig { x, x_sq, s_mul }
    }

    fn synthesize(
        &self,
        config: MulConfig,
        mut layouter: impl Layouter<Fr>,
    ) -> Result<(), FrontendError> { 
        layouter.assign_region(
            || "mul_bench",
            |mut region| {
                for (i, xi) in self.xs.iter().enumerate() {
                    config.s_mul.enable(&mut region, i)?;
                    region.assign_advice(|| "x",    config.x,    i, || *xi)?;
                    region.assign_advice(|| "x_sq", config.x_sq, i, || xi.map(|v| v * v))?;
                }
                Ok(())
            },
        )
    }
}

// ── Utilitaires ───────────────────────────────────────────────────────────────
fn fmt_d(d: Duration) -> String {
    let s = d.as_secs_f64();
    if s < 1.0 { format!("{:.1} ms", s * 1e3) } else { format!("{:.3} s", s) }
}

fn fmt_b(n: usize) -> String {
    if n < 1024 { format!("{} B", n) }
    else if n < 1 << 20 { format!("{:.2} KiB", n as f64 / 1024.0) }
    else { format!("{:.2} MiB", n as f64 / (1024.0 * 1024.0)) }
}

fn bar(c: char, n: usize) { println!("{}", std::iter::repeat(c).take(n).collect::<String>()); }

// ── Main ─────────────────────────────────────────────────────────────────────
fn main() {
    let k: u32 = env::args().nth(1).and_then(|s| s.parse().ok()).unwrap_or(DEFAULT_K);
    let n_dom  = 1usize << k;
    let n_cst  = n_dom - BLINDING_ROWS;

    bar('═', 66);
    println!("  BENCHMARK HALO2/PLONK (PSE Monorepo Split) — BN254");
    bar('═', 66);
    println!("  k = {}  →  2^k = {}  →  Contraintes ≈ {}", k, n_dom, n_cst);
    println!();

    let mut timings: Vec<(&str, Duration)> = vec![];

    // 1. SRS
    bar('─', 66);
    println!("[1/5] Génération SRS (ParamsKZG) …");
    let t = Instant::now();
    let params: ParamsKZG<Bn256> = ParamsKZG::new(k);
    let d = t.elapsed();
    println!("  ✓ {}", fmt_d(d));
    timings.push(("Setup SRS", d));

    // 2. Témoin
    bar('─', 66);
    println!("[2/5] Génération témoin …");
    let t = Instant::now();
    let xs: Vec<Value<Fr>> = (0..n_cst).map(|_| Value::known(Fr::random(OsRng))).collect();
    let circuit = MulBenchCircuit { xs };
    let d = t.elapsed();
    println!("  ✓ {}", fmt_d(d));
    timings.push(("Génération témoin", d));

    // 3. Clés
    bar('─', 66);
    println!("[3/5] Génération des clés (VK & PK) …");
    let t = Instant::now();
    let vk = keygen_vk(&params, &circuit.without_witnesses()).expect("keygen_vk failed");
    let dvk = t.elapsed();
    let t = Instant::now();
    let pk = keygen_pk(&params, vk, &circuit.without_witnesses()).expect("keygen_pk failed");
    let dpk = t.elapsed();
    println!("  ✓ vk {} | pk {}", fmt_d(dvk), fmt_d(dpk));
    timings.push(("keygen_vk", dvk));
    timings.push(("keygen_pk", dpk));

    let instances: Vec<Vec<Fr>> = vec![];

    // 4. Preuve
    bar('─', 66);
    println!("[4/5] Génération de la preuve (create_proof) …");
    let t = Instant::now();
    let mut tw = Blake2bWrite::<Vec<u8>, G1Affine, Challenge255<G1Affine>>::init(vec![]);
    
    create_proof::<
        KZGCommitmentScheme<Bn256>,
        ProverSHPLONK<'_, Bn256>,
        Challenge255<G1Affine>,
        OsRng,
        Blake2bWrite<Vec<u8>, G1Affine, Challenge255<G1Affine>>,
        MulBenchCircuit,
    >(&params, &pk, &[circuit], &[instances.clone()], OsRng, &mut tw)
    .expect("create_proof failed");
    let proof   = tw.finalize();
    let dprove  = t.elapsed();
    let plen    = proof.len();
    println!("  ✓ {}  |  Taille : {}", fmt_d(dprove), fmt_b(plen));
    timings.push(("Génération preuve", dprove));

    // 5. Vérification
    bar('─', 66);
    println!("[5/5] Vérification de la preuve (verify_proof) …");
    let t = Instant::now();
    
    let verifier_params = params.verifier_params();
    let mut tr   = Blake2bRead::<&[u8], G1Affine, Challenge255<G1Affine>>::init(&proof[..]);
    
    // Appel à l'API 4 arguments sans durée de vie générique, renvoyant un booléen direct
    let is_valid = verify_proof::<
        KZGCommitmentScheme<Bn256>,
        VerifierSHPLONK<Bn256>,
        Challenge255<G1Affine>,
        Blake2bRead<&[u8], G1Affine, Challenge255<G1Affine>>,
        SingleStrategy<Bn256>,
    >(&verifier_params, pk.get_vk(), instances, &mut tr);
    
    assert!(is_valid, "VÉRIFICATION ÉCHOUÉE — preuve invalide !");
    
    let dverify = t.elapsed();
    println!("  ✓ VALIDE  {}", fmt_d(dverify));
    timings.push(("Vérification", dverify));

    // Résultats finaux
    println!();
    bar('═', 66);
    println!("  RÉSULTATS FINAUX   k={k}");
    bar('═', 66);
    for (label, dur) in &timings {
        println!("  {:<32} {:>10}", label, fmt_d(*dur));
    }
    println!("  {:<32} {:>10}", "Taille de la preuve", fmt_b(plen));
    bar('═', 66);
}
